1. Display customer name and cities from which they have placed orders

db.orders.aggregate({$group:{_id:{City:"$City",'Customer Name':'$Customer Name'}}},
{$project:{City:"$_id.City",'Customer Name':'$_id.Customer Name',_id:0}})


2. display any 5 records from a collection

db.orders.find().limit(5)


3. select 10 customer name, region and sales sort region in ascending orders and sales in descending order

db.orders.aggregate({$group:{_id:{cn:"$Customer Name",re:"$Region"},sa:{$sum:"$Sales"}}},
                    {$project:{'Customer Name':"$_id.cn",Region:"$_id.re",Sales:"$sa",_id:0}},
                    {$sort:{Region:1,Sales:-1}},{$limit:10})

4. display customer name of 5 customers who are from central region based on the maximum profit they have given

db.orders.aggregate({$match:{Region:"Central"}},
		{$group:{_id:"$Customer Name",pr:{$sum:"$Profit"}}},
		{$sort:{pr:-1}},{$limit:5},{$project:{'Customer Name':"$_id",_id:0}})


5. display customer name of customers who have opted for "L" priority

db.orders.aggregate({$match:{Priority:"L"}},{$group:{_id:"$Customer Name"}},{$project:{'Customer Name':"$_id",_id:0}})


6. display customer name, region and city of customers who have not opted for priority

db.orders.find({$and:[{Priority:{$exists:false}},{Priority:{$eq:null}}]},{'Customer Name':1,Region:1,City:1,_id:0})


7. display number of customers who have opted for "H" priority and color is red

db.orders.find({Priority:"H",Color:"Red"}).count()


8. display name of all cities from where orders have been placed for chains

db.orders.aggregate({$match:{Subcategory:"Chains"}},
                    {$group:{_id:"$City"}},{$project:{City:"$_id",_id:0}})


9. display number of unique orders placed

db.orders.aggregate({$group:{_id:"$OrderID"}},{$count:"Unique Orders Count"})


10. display states where silver color is preferred

db.orders.aggregate({$match:{Color:"Silver"}},{$group:{_id:"$State"}},{$project:{State:"$_id",_id:0}})


11. display customer name of all customers, whose name starts with letter A irrespective of case

db.orders.aggregate({$match:{'Customer Name':{$regex:/^A/i}}},
                    {$group:{_id:"$Customer Name"}},
                    {$project:{"Customer Name":"$_id",_id:0}})


12. display customer name of all customers who are from Texas or New York city

db.orders.aggregate({$match:{$or:[{State:"Texas"},{City:"New York City"}]}},
                    {$group:{_id:"$Customer Name"}},
                    {$project:{"Customer Name":"$_id",_id:0}})


13. display productid where quantity is between 2 and 5(inclusive)

db.orders.aggregate({$match:{$and:[{qty:{$gte:2}},{qty:{$lte:5}}]}},
                    {$group:{_id:"$Product ID"}},
                    {$project:{'Product ID':"$_id",_id:0}})


14. add a column unit price by computing sales/quantity

db.orders.updateMany({},[{$set:{'Unit Price':{$divide:["$Sales","$qty"]}}}])

17. display number of product names under home office segment

18. display customer name, order number, order date from east who bought handlebars

19. display profit made by corporate segment

                    {$group:{_id:null,'Total Profit':{$sum:"$Profit"}}},
                    {$project:{_id:0,'Total Profit':1}})


20. display total quantity of sub category mountain bikes
