1. Display total number of orders placed by each customer

db.orders.aggregate({
$group:{_id:"$Customer Name",NumberOfOrders:{$sum:1}}},
        {$project:{CustomerName:"$_id",_id:0,NumberOfOrders:1}}
)


2. calculate average order value of each customer

db.orders.aggregate([{$group:{_id:"$Customer Name",totalSales:{$sum: "$Sales"},orderCount:{$sum: 1}}},
 {$project:{_id:0,customerName:"$_id",averageOrderValue:{$divide:["$totalSales","$orderCount"]}}}])


3. display total revenue and profit generated by each product

db.orders.aggregate({
$group:{_id:"$Product ID",sos:{$sum:"$Sales"},sop:{$sum:"$Profit"}}
},
                    {$project:{ProductID:"$_id",_id:0,TotalRevenue:"$sos",Profit:"$sop"}})


4. identify the products which was sold more

db.orders.aggregate({
$group:{_id:"$Product ID",NoOfTimesSold:{$sum:1}}
},{$sort:{NoOfTimesSold:-1}},{$project:{ProductID:"$_id",_id:0,NoOfTimesSold:1}})


5. list the total number of products who bought from different region

db.orders.aggregate({
$group:{_id:{na:"$Customer Name",re:"$Region"}}
},{$group:{_id:"$_id.re",Total:{$sum:1}}},
                    {$project:{_id:0,Region:"$_id",TotalNoOfCustomers:"$Total"}})


6. what is the average profit from east region from corporate segment customers

db.orders.aggregate([
  {$match:{Region:"East",Segment:"Corporate"}},
  {$group:{_id:null,su:{$sum:"$Profit"},to:{$sum:1}}},
  {$project:{_id:0,AverageProfit:{$divide:["$su","$to"]}}}
])


7. list the customers who have given repeat business

db.orders.aggregate([
  {$group:{_id:{cn:"$Customer Name",ca:"$Category"},co:{$sum:1}}},
  {$match:{co:{$gt:1}}},
  {$group:{_id:"$_id.cn"}},
  {$project:{_id:0,CustomerName:"$_id"}}
])


8. list the details of a product its category & subcategory which has been bought by maximum customers

db.orders.aggregate([{$group:{_id:{pi:"$Product ID",ca:"$Category",sc:"$Subcategory"},to:{$sum:1}}},
                     {$sort:{to:-1}},
                     {$project:{_id:0,ProductID:"$_id.pi",Category:"$_id.ca",Subcategory:"$_.id.sc"}}])


9. which region and which segment are contributing to minimum profit

db.orders.aggregate([{
$group:{_id:{re:"$Region",se:"$Segment",pr:"$Profit"}}

                     {$sort:{pr:1}},
                     {$project:{_id:0,Region:"$_id.re",Segment:"$_id.se"}},
                     {$limit:1}])
